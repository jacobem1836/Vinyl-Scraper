    ```python
    from fastapi import APIRouter, Depends, Form, Request
    from fastapi.responses import RedirectResponse
    from fastapi.templating import Jinja2Templates
    from sqlalchemy.orm import Session

    from app.auth import hash_password, verify_password
    from app.database import get_db
    from app.models import User

    auth_router = APIRouter(tags=["auth"])
    templates = Jinja2Templates(directory="templates")

    MIN_PASSWORD_LEN = 8


    def _safe_next(next_param: str | None) -> str:
        """Whitelist next= to relative paths only. Prevents open-redirect."""
        if not next_param:
            return "/"
        if not next_param.startswith("/") or next_param.startswith("//"):
            return "/"
        return next_param


    @auth_router.get("/login")
    async def login_page(request: Request, next: str = "", error: str = ""):
        return templates.TemplateResponse(
            "login.html",
            {"request": request, "next": next, "error": error, "email": ""},
        )


    @auth_router.post("/login")
    async def login_submit(
        request: Request,
        email: str = Form(...),
        password: str = Form(...),
        next: str = Form(""),
        db: Session = Depends(get_db),
    ):
        email_norm = email.strip().lower()
        user = db.query(User).filter_by(email=email_norm).first()
        if user is None or not verify_password(password, user.password_hash):
            return templates.TemplateResponse(
                "login.html",
                {
                    "request": request,
                    "next": next,
                    "error": "Invalid email or password.",
                    "email": email_norm,
                },
                status_code=200,
            )
        request.session["user_id"] = user.id
        return RedirectResponse(url=_safe_next(next), status_code=303)


    @auth_router.get("/signup")
    async def signup_page(request: Request, next: str = "", error: str = ""):
        return templates.TemplateResponse(
            "signup.html",
            {"request": request, "next": next, "error": error, "email": ""},
        )


    @auth_router.post("/signup")
    async def signup_submit(
        request: Request,
        email: str = Form(...),
        password: str = Form(...),
        next: str = Form(""),
        db: Session = Depends(get_db),
    ):
        email_norm = email.strip().lower()

        def _err(msg: str):
            return templates.TemplateResponse(
                "signup.html",
                {"request": request, "next": next, "error": msg, "email": email_norm},
                status_code=200,
            )

        if "@" not in email_norm or "." not in email_norm:
            return _err("Please enter a valid email address.")
        if len(password) < MIN_PASSWORD_LEN:
            return _err(f"Password must be at least {MIN_PASSWORD_LEN} characters.")

        existing = db.query(User).filter_by(email=email_norm).first()
        if existing is not None:
            return _err("An account with that email already exists.")

        user = User(email=email_norm, password_hash=hash_password(password))
        db.add(user)
        db.commit()
        db.refresh(user)
        request.session["user_id"] = user.id
        return RedirectResponse(url=_safe_next(next), status_code=303)


    @auth_router.post("/logout")
    async def logout(request: Request):
        request.session.clear()
        return RedirectResponse(url="/login", status_code=303)
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → /login, /signup | User-supplied email and password cross here. |
| /signup → DB | New User row written; password must be hashed. |
| /login → session | Session cookie set with user_id. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-29-07 | Tampering | `next` query/form parameter (open redirect) | mitigate | `_safe_next()` whitelists relative paths starting with a single `/`. External URLs and protocol-relative `//host` are rejected. |
| T-29-08 | Information Disclosure | login error message reveals which field was wrong | mitigate | Single generic "Invalid email or password" used for both unknown-email and wrong-password cases. |
| T-29-09 | Information Disclosure | password value re-rendered in error response | mitigate | Templates re-render `email` only; `password` value is never round-tripped. |
| T-29-10 | Information Disclosure | passwords logged via uvicorn access logs as form bodies | accept | Standard uvicorn does not log POST bodies. No additional logging added. |
| T-29-11 | Spoofing | brute-force login attempts | accept (Phase 29) | Rate limiting is Phase 31 SEC-02. Documented as deferred. |
| T-29-12 | Tampering / CSRF on /signup, /login, /logout | mitigate | Starlette SessionMiddleware sets `same_site=lax`, which blocks cross-site POSTs from third-party origins for the cookie. POSTs are accepted only from same-origin form submissions in Phase 29. |
| T-29-13 | Information Disclosure | bcrypt hash exposed via API serialization | mitigate | No User schema is added to `app/schemas.py`; the User model is never returned via Pydantic response_model. |
</threat_model>
