FROM python:3.11-slim

WORKDIR /app

# System deps for psycopg2-binary and httpx
RUN apt-get update && apt-get install -y --no-install-recommends \
    gcc \
    libpq-dev \
    && rm -rf /var/lib/apt/lists/*

# Install Python deps first for layer caching
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy application
COPY app ./app
COPY static ./static
COPY templates ./templates

EXPOSE 8080

CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8080"]
```

Notes:
- Port 8080 (not ${PORT}) per Fly.io convention from CONTEXT D-04
- No multi-stage build per D-04 ("overkill for a personal project")

  auto_stop_machines = false
  auto_start_machines = true
  min_machines_running = 1
  processes = ["app"]

[[http_service.checks]]
  interval = "30s"
  timeout = "5s"
  grace_period = "10s"
  method = "GET"
  path = "/api/health"

[[vm]]
  cpu_kind = "shared"
  cpus = 1
  memory_mb = 256
```

Key choices (per CONTEXT "Claude's Discretion"):
- `primary_region = "syd"` — closest to Brisbane, AU
- `auto_stop_machines = false` and `min_machines_running = 1` — APScheduler MUST stay alive (per CONTEXT integration points)
- `internal_port = 8080` — matches Dockerfile EXPOSE
- `/api/health` healthcheck — reuses existing endpoint (per CONTEXT integration points)
- 256 MB RAM, 1 shared CPU — within Fly free tier (3 shared-cpu-1x VMs free)

Note: `app = "crate"` is a placeholder — Plan 03 will run `flyctl launch` which may rewrite this if a name conflict exists.
